mcu: STM32H755II
package: UFBGA176

port clock:
  channel IN
  channel OUT
  color "brown"
  
  config "XTAL":
    IN = RCC_OSCIN
    OUT = RCC_OSCOUT
  
port debug:
  channel DIO = *_SWDIO
  channel CLK = *_SWCLK
  channel SWO = *_SWO
  channel TX = USART*_TX $u
  channel RX = USART*_RX $u
  channel DP = USB*_OTGFSDP
  channel DM = USB*_OTGFSDM
  require dma(RX, "USART_RX")
  require dma(TX, "USART_TX")
  color "brown"

port SAFETY:
  channel TX = USART*_TX $u
  channel RX = USART*_RX $u
  require dma(RX, "USART_RX")
  require dma(TX, "USART_TX")

port ECAT_GD:
  channel miso = SPI*_MISO $s
  channel mosi = SPI*_MOSI $s
  channel sck = SPI*_SCK $s
  channel cs_ecat = SPI*_NSS $s
  channel cs_gd = OUT
  channel sync0 = IN
  channel sync1 = OUT
  channel irq = IN
  require dma(miso, "SPI_RX")
  require dma(mosi, "SPI_TX")

port ENC0:
  channel miso
  channel mosi
  channel sck
  channel cs
  channel miso_tx_en
  channel mosi_tx_en
  channel sck_tx_en
  color "orange"

  config "SPI":
    miso = SPI*_MISO $s
    mosi = SPI*_MOSI $s
    sck = SPI*_SCK $s
    cs = SPI*_NSS $s
    miso_tx_en = OUT
    mosi_tx_en = OUT
    sck_tx_en = OUT
    require dma(miso, "SPI_RX")
    require dma(mosi, "SPI_TX")
    
  config "TIM":
    miso = TIM*_CH[1,2] $t
    mosi = TIM*_CH[3,4] $t
    sck = TIM*_CH[3,4] $t
    
  config "UART":
    miso = USART*_RX $u
    mosi = USART*_TX $u
    require dma(miso, "USART_RX")
    require dma(mosi, "USART_TX")
    
port ENC1 from ENC0:
  color "green"

port ENC2 from ENC0:
  color "red"
 
port MOT:
  # pwm
  channel A = TIM[1,8,20]_CH1 $t
  channel AN = TIM[1,8,20]_CH1N $t
  channel B = TIM[1,8,20]_CH2 $t
  channel BN = TIM[1,8,20]_CH2N $t
  channel C = TIM[1,8,20]_CH3 $t
  channel CN = TIM[1,8,20]_CH3N $t
  channel BRAKE = TIM[1,8,20]_CH4 $t
  channel FAN = TIM[2-5]_CH*
  
  # phase currents + voltages
  channel IA = ADC1_INP[1-3]
  channel IB = ADC2_INP[1-3]
  channel IC = ADC3_INP[1-3]
  channel UA = ADC1_INP[1-6]
  channel UB = ADC2_INP[1-6]
  channel UC = ADC3_INP[1-6]
  
  #dc current + voltage
  channel IDC = ADC1_INP[7-9]
  channel UDC = ADC2_INP[7-9]
  
  # brake current
  channel IBRAKE = ADC1_INP[7-9]
  
  # motor + mosfet temperature
  channel TMOT_A = ADC1_INP[10-15]
  channel TMOT_B = ADC2_INP[10-15]
  channel TMOT_C = ADC3_INP[10-15]
  channel TFET = ADC3_INP[10-15]
  channel THEATSINK = ADC3_INP[10-15]
  
  channel STO0 = TIM[1,8,20]_BKIN $t
  channel STO1 = TIM[1,8,20]_BKIN2 $t

  require dma(IA)
  require dma(IB)
  require dma(IC)
  color "blue"
