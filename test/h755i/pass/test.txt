port ENC0:
  channel MA
  channel MO
  channel SL
  channel CS
  color "orange"

  config "SPI":
    SL = SPI*_MISO $s
    MA = SPI*_SCK $s
    MO = SPI*_MOSI $s # reserved for future use
    CS = SPI*_NSS $s # reserved for future use
    require dma(SL, "SPI_RX")
    
  # optional
  config "UART": 
    SL = USART*_RX $u # reserved for future use, connected
    MA = USART*_TX $u # reserved for future use, connected
    require dma(MA, "USART_TX")
    require dma(SL, "USART_RX")


port ENC1 from ENC0:
  color "green"
  
port ENC2 from ENC0:
  channel MO
  channel MA_txen
  channel MO_txen
  color "red"
  
  config "SPI":
    SL = SPI*_MISO $s
    MO = SPI*_MOSI $s
    MA = SPI*_SCK $s
    MO_txen = OUT
    require dma(MO, "SPI_TX")
    require dma(SL, "SPI_RX")
  
  config "UART":
    SL = USART*_RX $u
    MA = USART*_TX $u
    MA_txen = USART*_DE $u
    MO_txen = OUT
    require dma(MA, "USART_TX")
    require dma(SL, "USART_RX")
    
  config "TIM":
    MO = TIM[1,2,3,4,5,8,20]_CH[1,2] $t
    SL = TIM*_CH[1,2] $t
    MA = TIM*_CH[3,4] $t
    MO_txen = OUT
    MA_txen = OUT

  channel CN = TIM[1,8,20]_CH3N $t
  channel BRAKE = TIM[1,8,20]_CH4 $t
  channel FAN = TIM*_CH*
  
  # alternative mapping for hrtim
